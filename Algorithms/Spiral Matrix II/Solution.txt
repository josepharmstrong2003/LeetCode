class Solution:
    def generateMatrix(self, n: int) -> List[List[int]]:
        grid = []
        for x in range(0, n):
            toAdd = []
            for y in range(0, n):
                toAdd.append(" ")
            grid.append(toAdd)

        xMin = 0 
        xMax = n-1
        yMin = 0
        yMax = n-1
        x = 0
        y = 0
        direction = "Right"
        for i in range(1, n*n+1):
            
            grid[y][x] = i 
            
          
            if direction == "Right":
                
                
                x = x + 1 
                if x == xMax:
                    
                    yMin += 1
                    direction = "Down"

            elif direction == "Down":
                
                y = y + 1 
                if y == yMax:
                    xMax -= 1
                    direction = "Left"
            elif direction == "Left":
                
                x = x - 1 
                if x == xMin:
                    yMax -= 1
                    direction = "Up"
            elif direction == "Up":
                
                y = y - 1 
                if y == yMin:
                    xMin += 1
                    direction = "Right"
            
        return(grid)
